# Image Classification Plugin

# proto
V1

```
syntax = "proto3";

package docai_img_clf;
option go_package = "./;docai_img_clf";

message ImageClfRequest {
    repeated bytes image_bytes = 1;  // Input images
}

message ClfResult {
    string label = 1;       // Predicted category label
    float proba = 2;        // Probability of the category
}

message ImageClfResponse {
    repeated ClfResult results = 1; 
    string Status = 2;      // Empty if success; error message if failed
}

service ImageClfService {
    rpc ImageClf(ImageClfRequest) returns (ImageClfResponse);
}
```
V2

```
syntax = "proto3";

package docai_img_clf;
option go_package = "./;docai_img_clf";

message ImageClfRequest {
    repeated bytes image_bytes = 1;  // Input images
    string image_source = 2;         // Source of the image (e.g., file, frame)
    string extra = 3;                // Extra parameters
}

message ClfResult {
    string label = 1;       // Predicted category label
    float proba = 2;        // Probability of the category
}

message ImageClfResponse {
    repeated ClfResult results = 1; 
    string Status = 2;      // Empty if success; error message if failed
}

service ImageClfService {
    rpc ImageClf(ImageClfRequest) returns (ImageClfResponse);
}
```
